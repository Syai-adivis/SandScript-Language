cmake_minimum_required(VERSION 3.16)
project(SandScript VERSION 3.1.3 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SAND_ENABLE_QT_WIDGETS "Enable Qt6 Widgets support for Sand widget declarations" ON)
option(SAND_ENABLE_QT_NETWORK "Enable Qt6 Network support for HTTP requests" ON)

set(HEADERS RuntimeValue.h analyzer.h includes.h Interpreter.h prepross.h SetColor.h qt_widget_runtime.h qt_network_runtime.h SandPlugin.h)
set(COMMON_SRCS prepross.cpp)
set(TARGETS QSandi QSandShell)
add_executable(QSandi Sandi.cpp ${COMMON_SRCS})
add_executable(QSandShell SandShell.cpp ${COMMON_SRCS})

if(SAND_ENABLE_QT_WIDGETS)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    foreach(target IN LISTS TARGETS)
        target_sources(${target} PRIVATE qt_widget_runtime.h)
        target_compile_definitions(${target} PRIVATE SAND_ENABLE_QT_WIDGETS)
        target_link_libraries(${target} PRIVATE Qt6::Widgets)
    endforeach()

    find_package(Qt6 QUIET COMPONENTS OpenGLWidgets)
    if(TARGET Qt6::OpenGLWidgets)
        foreach(target IN LISTS TARGETS)
            target_compile_definitions(${target} PRIVATE SAND_ENABLE_QT_OPENGL_WIDGETS)
            target_link_libraries(${target} PRIVATE Qt6::OpenGLWidgets)
        endforeach()
    endif()

    find_package(Qt6 QUIET COMPONENTS QuickWidgets)
    if(TARGET Qt6::QuickWidgets)
        foreach(target IN LISTS TARGETS)
            target_compile_definitions(${target} PRIVATE SAND_ENABLE_QT_QUICK_WIDGETS)
            target_link_libraries(${target} PRIVATE Qt6::QuickWidgets)
        endforeach()
    endif()
endif()

if(SAND_ENABLE_QT_NETWORK)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    foreach(target IN LISTS TARGETS)
        target_compile_definitions(${target} PRIVATE SAND_ENABLE_QT_NETWORK)
        target_link_libraries(${target} PRIVATE Qt6::Network)
    endforeach()
endif()

if(MSVC)
    target_compile_options(QSandi PRIVATE /W4 /O2)
    target_compile_options(QSandShell PRIVATE /W4 /O2)
else()
    target_compile_options(QSandi PRIVATE -Wall -Wextra -O2)
    target_compile_options(QSandShell PRIVATE -Wall -Wextra -O2)
endif()
